-- 1. VERIFICAR Y BORRAR LA BASE DE DATOS SI YA EXISTE
IF EXISTS (SELECT name FROM sys.databases WHERE name = N'GESTOR')
BEGIN
    -- Cierra cualquier conexión activa para poder borrarla sin errores
    ALTER DATABASE GESTOR SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
    DROP DATABASE GESTOR;
END
GO

-- 2. CREAR LA BASE DE DATOS LIMPIA
CREATE DATABASE GESTOR;
GO

USE GESTOR;
GO

-- 3. CREACIÓN DE TABLAS MAESTRAS (Sin dependencias)

CREATE TABLE Cliente (
    ID INT PRIMARY KEY, -- Ingresado manualmente (DNI)
    Nombre VARCHAR(150) NOT NULL,
    Direccion VARCHAR(200),
    Correo VARCHAR(100),
    NumeroContacto VARCHAR(15)
)
GO

CREATE TABLE Proveedor (
    RUC VARCHAR(11) PRIMARY KEY, -- Ingresado manualmente (RUC)
    RazonSocial VARCHAR(150) NOT NULL,
    Contacto VARCHAR(100),
    Direccion VARCHAR(200),
    NumeroContacto VARCHAR(15),
    CorreoContacto VARCHAR(100)
)
GO

CREATE TABLE Producto (
    ID INT PRIMARY KEY IDENTITY(100000, 1), -- Autoincrementable
    Descripcion VARCHAR(150) NOT NULL,
    PrecioVentaBase DECIMAL(10, 2),
    PrecioCompra DECIMAL(10, 2),
    Stock INT DEFAULT 0
)
GO

-- 4. CREACIÓN DE TABLAS TRANSACCIONALES (Dependen de las maestras)

CREATE TABLE Pedido (
    Codigo INT PRIMARY KEY IDENTITY(200000, 1), -- Autoincrementable
    Cliente_ID INT NOT NULL REFERENCES Cliente(ID),
    FechaEmision DATE NOT NULL,
    FechaRecepcion DATE,
    Estado VARCHAR(50) NOT NULL,
    TipoVenta VARCHAR(50) NOT NULL,
    FechaLimitePago DATE,
    DeudaPendiente DECIMAL(10, 2) DEFAULT 0.00
)
GO

CREATE TABLE OrdenCompra (
    Codigo INT PRIMARY KEY, -- Calculado con MAX(Codigo) + 1 en Java
    Proveedor_RUC VARCHAR(11) NOT NULL REFERENCES Proveedor(RUC),
    FechaIngreso DATE NOT NULL,
    Estado VARCHAR(50) NOT NULL
)
GO

-- 5. CREACIÓN DE TABLAS DE DETALLES (Llaves compuestas)

CREATE TABLE DetallePedido (
    Pedido_Codigo INT NOT NULL REFERENCES Pedido(Codigo),
    Producto_ID INT NOT NULL REFERENCES Producto(ID),
    CantidadVendida INT NOT NULL,
    PrecioVentaCongelado DECIMAL(10, 2) NOT NULL,
    PRIMARY KEY (Pedido_Codigo, Producto_ID)
)
GO

CREATE TABLE LoteProducto (
    OrdenCompra_Codigo INT NOT NULL REFERENCES OrdenCompra(Codigo),
    Producto_ID INT NOT NULL REFERENCES Producto(ID),
    CantidadIngresada INT NOT NULL,
    PrecioCompraIndividual DECIMAL(10, 2) NOT NULL,
    PRIMARY KEY (OrdenCompra_Codigo, Producto_ID)
)
GO

-- 6. CREACIÓN DE TABLAS DE HISTORIAL Y FINANZAS

CREATE TABLE PagoAbono (
    CodigoRecibo INT PRIMARY KEY IDENTITY(400000, 1), -- Autoincrementable
    Pedido_Codigo INT NOT NULL REFERENCES Pedido(Codigo),
    FechaAbono DATE NOT NULL,
    MontoAbonado DECIMAL(10, 2) NOT NULL
)
GO

CREATE TABLE MovimientosKardex (
    IdMovimiento INT PRIMARY KEY IDENTITY(100000, 1), -- Autoincrementable
    Producto_ID INT NOT NULL REFERENCES Producto(ID),
    FechaMovimiento DATE NOT NULL,
    TipoMovimiento VARCHAR(50) NOT NULL,
    CantidadFisica INT NOT NULL,
    ValorUnitario DECIMAL(10, 2) NOT NULL,
    SaldoCantidadActual INT NOT NULL,
    SaldoCostoPromedio DECIMAL(10, 2) NOT NULL
)
GO
