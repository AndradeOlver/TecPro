package DAO;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;

public class ConexionSQL {
    // 1. Parámetros de Ruta 
    private static final String SERVIDOR = "localhost";
    private static final String PUERTO = "1433";
    private static final String BASE_DATOS = "GESTOR";
    
    // 2. Se agrega integratedSecurity=true (Autenticación de Windows)
    private static final String URL = "jdbc:sqlserver://" + SERVIDOR + ":" + PUERTO + 
                                      ";databaseName=" + BASE_DATOS + 
                                      ";integratedSecurity=true;encrypt=true;trustServerCertificate=true;";

    public static Connection probarConexion() {
        Connection conexion = null;
        
        try {
            Class.forName("com.microsoft.sqlserver.jdbc.SQLServerDriver");
            
            // 3. Se pasa ÚNICAMENTE la URL (Se eliminan las variables de usuario y clave)
            conexion = DriverManager.getConnection(URL);
            
            if (conexion != null) {
                System.out.println("¡Conexión establecida con éxito mediante Autenticación de Windows!");
            }
            
        } catch (ClassNotFoundException e) {
            System.err.println("Error crítico: No se encontró el Driver de SQL Server.");
            System.err.println("Detalle: " + e.getMessage());
        } catch (SQLException e) {
            System.err.println("Error de conexión: Verifica configuración o la librería DLL.");
            System.err.println("Detalle: " + e.getMessage());
        }
        
        return conexion;
    }

    public static void cerrarConexion(Connection conexion) {
        if (conexion != null) {
            try {
                conexion.close();
                System.out.println("Conexión cerrada correctamente.");
            } catch (SQLException e) {
                System.err.println("Error al intentar cerrar la conexión: " + e.getMessage());
            }
        }
    }
}
